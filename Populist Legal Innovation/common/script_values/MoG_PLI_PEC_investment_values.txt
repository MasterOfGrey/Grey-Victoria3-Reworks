### INVESTMENT ###
# Additions from PEC laws (investment)
mog_pli_pec_investment_add = {
	if = {
		limit = { PEC_is_active = yes }
		value = 0
		add = mog_pli_pec_investment_bank
	}
	else = {
		value = 0
	}
}

mog_pli_pec_investment_bank = {
	if = {
		limit = { has_law_or_variant = law_type:law_pec_merchant_banks }
		add = 5
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_pec_free_banking }
		add = 20
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_pec_central_bank_metallic }
		add = 10
		add = {
			add = 5	# Eventually exceeds free banking
			multiply = {
				value = institution:institution_pec_central_bank.investment
			}
		}
	}
	else_if = {
		limit = { has_law_or_variant = law_type:law_pec_abolish_money }
		add = 0		# good luck investing without money
	}
	else = {		# For random modded laws
		add = mog_pli_pec_investment_bank_modded
	}
}
mog_pli_pec_investment_bank_modded = 5
